int main ()
{
    stdio_init_all();

    leds_init();

    for(;;) {
        log(LogLevel::INFORMATION, "LED driver Lab 1");
        leds_set(0, 255, 0, 255);   // LED 0 pink
        leds_set(1, 255, 0, 255);   // LED 1 pink
        leds_set(2, 0, 0, 255);   // LED 2 blue (LED 2 is broken so blue + red = pink)
        leds_set(3, 255, 0, 255);   // LED 3 pink

        leds_set_range(4, 7, 0, 0, 255); // LED 4 to 7 blue

        leds_set_range(8, 11, 255, 0, 0); // LED 8 to 11 red
        leds_show();

        sleep_ms(1000);

        // Now change only LED 1
        leds_set(1, 0, 255, 0); // Change LED 1 to green
        leds_show();

        sleep_ms(1000);

        leds_set(0, 255, 0, 0);   // LED 0 red
        leds_set(1, 255, 0, 0);   // LED 1 red
        leds_set(2, 0, 0, 0);   // LED 2 off (becomes red)
        leds_set(3, 255, 0, 0);   // LED 3 red

        leds_set_range(4, 7, 255, 0, 255); // LED 4 to 7 pink

        leds_set_range(8, 11, 0, 0, 255); // LED 8 to 11 blue
        leds_show();

        sleep_ms(1000);

        leds_set(0, 0, 0, 255);   // LED 0 blue
        leds_set(1, 0, 0, 255);   // LED 1 blue
        leds_set(2, 0, 0, 0);   // LED 2 off (becomes red)
        leds_set(3, 0, 0, 255);   // LED 3 blue

        leds_set_range(4, 7, 255, 0, 0); // LED 4 to 7 red

        leds_set_range(8, 11, 255, 0, 255); // LED 8 to 11 pink
        leds_show();

        sleep_ms(1000);

        // Clear all LED's
        leds_clear();
        leds_show();

        sleep_ms(1000);
    }

    return 0;
}
