static void show_spirit_level(lis3dh_g_t g)
{
    leds_clear();

    if (g.x > 0.3f) {
        leds_set(0, 0, 255, 0);
    }
    else if (g.x < -0.3f) {
        leds_set(3, 0, 255, 0);
    }

    if (g.y > 0.3f) {
        leds_set(6, 0, 0, 255);
    } else if (g.y < -0.3f) {
        leds_set(9, 0, 0, 255);
    }

    if (fabsf(g.x) < 0.2f && fabsf(g.y) < 0.2f) {
        for (int i = 0; i < 12; i++) {
            leds_set(i, 0, 255, 0);
        }
    }

    leds_show();
}

int main()
{
    stdio_init_all();
    sleep_ms(2000);

    leds_init();

    printf("PROGRAM STARTED\r\n");

    leds_clear();
    for (int i = 0; i < 12; i++) {
        leds_set(i, 0, 0, 255); //blue
    }
    leds_show();
    sleep_ms(1000);

    if (!lis3dh_init()) {
        while (true) {
            printf("Accelerometer init failed\r\n");
            
            leds_clear();
            for (int i=0; 1 < 12; i++) {
                leds_set(i, 255, 0, 0); //all red = accel failed
            }
            leds_show();

            sleep_ms(500);

            leds_clear();
            leds_show();

            sleep_ms(500);
        }
    }

    while (true) {
        lis3dh_g_t g = lis3dh_read_g();

        printf("X: %.3fg, Y: %.3fg, Z: %.3fg\r\n", g.x, g.y, g.z);

        show_spirit_level(g);

        sleep_ms(100);
    }

    return 0;
}
